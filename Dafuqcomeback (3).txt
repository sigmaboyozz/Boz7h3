--// FUR INFECTION

    Default = false,
    Save = false,

            task.spawn(function()
                local success, err = pcall(function()
                    loadstring(game:HttpGet(

            -- =========================================

            -- =========================================

                local success, err = pcall(function()

                    ))()

                if not success then
                    warn("[119538320626412] Error:", err)
                end

            end

            -- =========================================
            -- PLACE ID 116454396851335
            -- =========================================

            if game.PlaceId == 116454396851335 then

                local success, err = pcall(function()
                    loadstring(game:HttpGet(
                        "https://raw.githubusercontent.com/Dreamer987/Auizhe/refs/heads/main/Arhurritan"
                    ))()
                end)

                    warn("[116454396851335] Error:", err)

            -- =========================================
            -- PLACE ID 3177438863
            -- =========================================

            if game.PlaceId == 3177438863 then

                local args = {
                    [1] = {
                        ["UniqueTag"] = "TeleporterGui",
                        ["Description"] = "Join World [Dungeon Hub]?",
                        ["LeftButton"] = "Join",
                        ["Timer"] = 30,
                        ["Prompt"] = "TeleportDirect",
                        ["RightButton"] = "Cancel",

                end

        end)

})
