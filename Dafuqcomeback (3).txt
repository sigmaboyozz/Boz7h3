--// AUTO SAVE TOGGLE
--// =========================================

local OrionLib = loadstring(game:HttpGet(
    "https://raw.githubusercontent.com/jensonhirst/Orion/main/source"
))()

local Window = OrionLib:MakeWindow({
    Name = "🔥 Main Scripts",
    HidePremium = false,
    SaveConfig = true,
    ConfigFolder = "MainScripts_AutoSave"
})

local Tab = Window:MakeTab({
    Name = "Main",
    Icon = "rbxassetid://4483345998",
    PremiumOnly = false
})

--// =========================================
--// FUR INFECTION
--// KHÔNG LƯU
--// =========================================

Tab:AddToggle({
    Name = "Fur Infection outbreak",
    Default = false,
    Save = false,
    Flag = "FurInfection",
    Callback = function(Value)

        if Value then
            task.spawn(function()
                local success, err = pcall(function()
                    loadstring(game:HttpGet(
                        "https://raw.githubusercontent.com/Dreamer987/Furryinfectionmetalbel/refs/heads/main/Furinfec"
                    ))()
                end)

                if not success then
                    warn("[Fur Infection] Error:", err)
                end
            end)
        end

    end
})
